    Public Function ObtenTCBySerie(ByVal Serie As String, ByVal fecha1 As String, fecha2 As String) As Double

        Dim tc As Double = 0

        Dim url As String = String.Format("https://www.banxico.org.mx/SieAPIRest/service/v1/series/{0}/datos/{1}/{2}", Serie, fecha1, fecha2)

        Dim client = New RestClient(url)
        Dim request = New RestRequest(Method.GET)

        request.AddHeader("Bmx-Token", "5395ac064dfc2abc4484d2d7b3ac5a793d3cdbdb63e099dfd18a4d4e02c20780")

        request.AddHeader("accept", "application/json")

        Dim response As IRestResponse = client.Execute(request)

        'Console.WriteLine(response.Content)

        Dim respuesta As JObject = JObject.Parse(response.Content)


        If response.IsSuccessful Then
            ' retorna los datos 
            'Return respuesta("id").ToString()
            Try
                For Each t As JToken In respuesta("bmx")("series")(0)("datos").Reverse()
                    'esto es valido 
                    'MsgBox(t("dato").ToString() & " Con " & t("fecha").ToString())
                    Try
                        tc = Double.Parse(t("dato").ToString())
                        Exit For
                    Catch ex As Exception
                    End Try
                Next
                'tc = Double.Parse(respuesta("bmx")("series")(0)("datos")(0)("dato").ToString())
            Catch ex As Exception
                tc = 0
            End Try
            Return Math.Round(tc, 2)
        Else
            Dim sg = respuesta("error")("detalle").ToString()
            tc = 0
            Throw New EvaluateException(sg)
        End If
    End Function
