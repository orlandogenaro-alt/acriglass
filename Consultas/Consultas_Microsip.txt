    Public Function retorna_historial_cambiaria(fecha_ini As Date, fecha_fin As Date) As DataTable
        Dim fbtran As FbTransaction
        fbtran = cnn_micro.BeginTransaction(IsolationLevel.Snapshot)
        Try
            Dim fbcom As New FbCommand("SELECT A.moneda_id,A.fecha, A.tipo_cambio FROM historia_cambiaria A WHERE A.fecha between @F_INI AND @F_FIN and A.moneda_id=87148 ", cnn_micro, fbtran)
            fbcom.Parameters.AddWithValue("F_INI", Format(fecha_ini, "dd.MM.yyyy"))
            fbcom.Parameters.AddWithValue("F_FIN", Format(fecha_fin, "dd.MM.yyyy"))
            Dim fbdat As New FbDataAdapter(fbcom)
            Dim tab As New DataTable
            fbdat.Fill(tab)
            fbtran.Commit()
            Return tab
        Catch ex As Exception
            fbtran.Rollback()
            MsgBox("retorna_historial_cambiaria " & ex.Message)
            Return Nothing
        End Try
    End Function
    Public Function consulta_cliente_por_id(ByVal id_cli As Integer) As DataTable
        Try
            Dim fbdat As New FbDataAdapter("select a.cliente_id,a.nombre,a.rfc,a.contacto,d.nombre as zonas_clientes, e.nombre as vendedor,a.telefono,
            a.email,b.nombre as cond_pago,c.nombre as tipo_cliente,a.calle,a.colonia,a.num_exterior,a.num_interior,
            a.poblacion,a.codigo_postal,f.nombre as ciudad,a.referencia,a.clave_cliente,a.CONDICIONES_COM,a.direccion
            ,coalesce(g.nombre,'') as sistema_met,coalesce(h.nombre,'') as moneda
            from clientes A
            left join condiciones_pago B on (a.cond_pago_id=b.cond_pago_id)
            left join tipos_clientes C on (a.tipo_cliente_id =c.tipo_cliente_id)
            left join zonas_clientes D on (d.zona_cliente_id=a.zona_cliente_id)
            left join vendedores E on (e.vendedor_id=a.vendedor_id)
            left join ciudades F on (f.ciudad_id=a.ciudad_id)
            left join sistema_dimenciones G on (g.sistema_dim_id=a.sistema_dim_id)
            left join tipos_moneda H on (h.moneda_id=a.moneda_id)
            where a.cliente_id='" & id_cli & "'", cnn_prod)
            Dim tab As New DataTable
            fbdat.Fill(tab)
            Return tab
        Catch ex As Exception
            Return Nothing
        End Try
    End Function

    Public Function retorna_datos_articulos_busqueda(texto_bus As String, tipo As Integer) As DataTable
        Dim sql As String = "select r.articulo_id,(select first 1 t.clave_articulo from claves_articulos T
        where t.articulo_id=r.articulo_id order by t.rol_clave_art_id) as clave_articulo , r.nombre,r.unidad_venta from articulos R where R.nombre containing @TEXTO_BUS"
        If tipo = 2 Then
            sql = "select a.articulo_id,a.clave_articulo, b.nombre,b.unidad_venta from claves_articulos A
            inner join articulos B on (a.articulo_id=b.articulo_id) where a.clave_articulo like @TEXTO_BUS"
        ElseIf tipo = 3 Then
            sql = "select a.articulo_id,a.clave_articulo, b.nombre,b.unidad_venta from claves_articulos A
            inner join articulos B on (a.articulo_id=b.articulo_id) where a.clave_articulo = @TEXTO_BUS"
        End If
        Dim fbtran As FbTransaction
        fbtran = cnn_micro.BeginTransaction(IsolationLevel.Snapshot)
        Try
            Dim fbcom As New FbCommand(sql, cnn_micro, fbtran)
            fbcom.Parameters.AddWithValue("TEXTO_BUS", texto_bus)
            Dim fbdat As New FbDataAdapter(fbcom)
            Dim tab As New DataTable
            fbdat.Fill(tab)
            fbtran.Commit()
            Return tab
        Catch ex As Exception
            fbtran.Rollback()
            MsgBox("retorna_datos_articulos " & ex.Message)
            Return Nothing
        End Try
    End Function

    Public Function retorna_datos_clientes(texto_bus As String, tipo As Integer) As DataTable

        If tipo = 2 Then

    Public Function retorna_monto_factura(folio As String) As DataTable
        Try
            Dim fbdat As New FbDataAdapter("select a.docto_cm_id,a.folio as folio_cm,a.importe_neto,a.tipo_cambio,b.docto_cp_id,b.folio,b.descripcion,c.nombre,(SELECT SUM(IMPORTE+IMPUESTO-IVA_RETENIDO-ISR_RETENIDO)
                                            FROM IMPORTES_DOCTOS_CP
                                            WHERE DOCTO_CP_ID = b.docto_cp_id) as importe_credito from doctos_cp B
                                            inner join doctos_entre_sis D on (d.docto_dest_id = b.docto_cp_id)
                                            inner join doctos_cm A on (a.docto_cm_id = d.docto_fte_id)
                                            inner join proveedores C on (c.proveedor_id =a.proveedor_id)
                                            where b.folio like '" & folio & "'", cnn_micro)
            Dim tab As New DataTable
            fbdat.Fill(tab)
            Return tab
        Catch ex As Exception
            Return Nothing
        End Try
    End Function
