    ''' <exception cref="ArgumentNullException"></exception>
    ''' <param name="articulos">DataTable que espera 4 valores -> Id del articulo, -> Cantidad de Articulos, -> Precio individual por Articulo, -> Descripción del Articulo</param>

    Public Function GuardaInvoiceQB(ByVal articulos As DataTable, ByVal id_cliente As String, ByVal invoice As Integer) As String
        'Dim invoice As Integer = retorna_invoice_id()
        Dim factura As String = "A" & invoice
        HayTokensAsync()

        Dim uri = String.Format("{0}/v3/company/{1}/invoice", baseURL, RelamId)
        System.Net.ServicePointManager.SecurityProtocol = SecurityProtocolType.Tls12

        Dim cliente = New RestClient(uri)
        Dim solicitud = New RestRequest(Method.POST)
        solicitud.AddHeader("Accept", "application/json")
        solicitud.AddHeader("Authorization", String.Concat("Bearer ", AccessToken))
        solicitud.AddHeader("Content-Type", "application/json")

        If articulos.Rows.Count < 1 Then
            Throw New Exception("No se han detectado articulos")
            Exit Function
        End If
        If articulos.Columns.Count <> 4 Then
            Throw New Exception("No las Columnas solicitados no son correctas")
            Exit Function
        End If
        'itemid  ---  cantidad  --- prtecio_unitario ---- descripciomn

        Dim lineObjct As New Collection()
        For Each art As DataRow In articulos.Rows
            lineObjct.Add(New With {.DetailType = "SalesItemLineDetail", .Amount = (art.ItemArray(1) * art.ItemArray(2)), .Description = art.ItemArray(3), .SalesItemLineDetail = New With {.ItemRef = New With {.value = art.ItemArray(0)}, .Qty = art.ItemArray(1), .UnitPrice = art.ItemArray(2)}})
        Next

        Dim jsonObject = New With {
                Key .DocNumber = factura,
                Key .Line = lineObjct,
                Key .CustomerRef = New With {
                    .value = id_cliente
                }
        }

        solicitud.AddJsonBody(jsonObject)

        Dim respuestas As IRestResponse = cliente.Execute(solicitud)

        Dim resp As JObject = JObject.Parse(respuestas.Content)
        Dim id_invoice As String = ""
        If respuestas.StatusCode = 200 Then
            'Console.WriteLine("Mensaje de quickbooks.")
            id_invoice = resp("Invoice")("Id").ToString()
            'Console.WriteLine(id_invoice)
            'incrementa_invoice_id(invoice + 1)
        Else
            Dim err = resp("Fault")("Error")(0)("Message").ToString()
            Dim detalle = resp("Fault")("Error")(0)("Detail").ToString()
            Throw New Exception(err & " Detalle: " & detalle)
            Return ""
        End If
        Return id_invoice
    End Function

        HayTokensAsync()

        Dim uri = String.Format("{0}/v3/company/{1}/query?query=select * from Item {2}", baseURL, RelamId, consulta)

        System.Net.ServicePointManager.SecurityProtocol = SecurityProtocolType.Tls12

        Dim cliente = New RestClient(uri)
        Dim solicitud = New RestRequest(Method.GET)
        solicitud.AddHeader("Accept", "application/json")
        solicitud.AddHeader("Authorization", String.Concat("Bearer ", AccessToken))
        solicitud.AddHeader("Content-Type", "text/plain")

            Dim respuestas As IRestResponse = cliente.Execute(solicitud)

            Dim articulos As New DataTable
            articulos.Columns.Add("Id")
            articulos.Columns.Add("Nombre")
            articulos.Columns.Add("Nombre Completo")

            articulos.Columns.Add("Precio Unitario")
            articulos.Columns.Add("Precio Compra")
            articulos.Columns.Add("Activo")
            articulos.Columns.Add("Escaso")

            Dim QueryResponse As JObject = JObject.Parse(respuestas.Content)

            If respuestas.StatusCode = 200 Then
                Dim items As JToken = QueryResponse("QueryResponse")("Item")
                For Each item As JToken In items

                    articulos.Rows.Add({item("Id").ToString(), item("Name").ToString(), item.SelectToken("FullyQualifiedName"), item.SelectToken("Description"), item.SelectToken("Type"), item.SelectToken("UnitPrice"), item.SelectToken("PurchaseCost"), item("Active").ToString(), item("sparse").ToString()})

            End If
            Return articulos
        Catch ex As Newtonsoft.Json.JsonReaderException
            MsgBox("error" & Convert.ToString(ex))
            Return Nothing
        End Try
    End Function

    Private Function RandomDataBase64url(lenigth As UInteger) As String
        Dim rng As RNGCryptoServiceProvider = New RNGCryptoServiceProvider()
        Dim bytes(lenigth) As Byte
        rng.GetBytes(bytes)

        Return Base64urlencodeNoPadding(bytes)
    End Function

    Private Function Base64urlencodeNoPadding(buffer As Byte()) As String
        Dim base64 As String = Convert.ToBase64String(buffer)
        base64 = base64.Replace("+", "-")
        base64 = base64.Replace("/", "_")
        base64 = base64.Replace("=", "")

        Return base64
    End Function

    Private Function sha256(input As String) As Byte()
        Dim bytes As Byte() = Encoding.ASCII.GetBytes(input)
        Dim sha As SHA256Managed = New SHA256Managed()
        Return sha.ComputeHash(bytes)
    End Function

    Private Function EncodeBase64ToString(ByVal valor As String) As String
        Return Convert.ToBase64String(Encoding.UTF8.GetBytes(valor))
    End Function

    Public Async Sub performCodeExcnahge(code As String, code_verifer As String, redicturi As String)

        ' creo que no llebaa el Barrirer
        Dim tokenRquestUri As String = TokienEndPoint

        Dim tokenReuestBody As String = String.Format("code={0}&redirect_uri={1}&client_id={2}&code_verifier={3}&client_secret={4}&scope=&grant_type=authorization_code",
                code,
                System.Uri.EscapeDataString(redicturi),
                ClientId,
                code_verifer,
                ClientSecret
                )
        ' enviar la respues
        Dim tokenRequest As HttpWebRequest = CType(WebRequest.Create(tokenRquestUri), HttpWebRequest)

    Private Sub GetTokens(ByVal auth_Code As String, ByVal reduri As String)

        Dim client = New RestClient(TokienEndPoint)

        Dim request = New RestRequest(Method.POST)
        request.AddHeader("Content-Type", "application/x-www-form-urlencoded")
        request.AddHeader("Authorization", String.Concat("Basic ", EncodeBase64ToString(ClientId + ":" + ClientSecret)))
        request.AddParameter("code", auth_Code, ParameterType.GetOrPost)
        request.AddParameter("redirect_uri", reduri, ParameterType.GetOrPost)
        request.AddParameter("grant_type", "authorization_code", ParameterType.GetOrPost)
        request.AddParameter("response_type", "aplicación/json", ParameterType.GetOrPost)

        Dim respuesta As IRestResponse = client.Execute(request)

        If respuesta.StatusCode = HttpStatusCode.OK Then

            Dim resp As JObject = JObject.Parse(respuesta.Content)

            IdToken = resp("id_token").ToString()
            RefreshToken = resp("refresh_token").ToString()
            AccessToken = resp("access_token").ToString()
            TokenType = resp("token_type").ToString()
            RefeshTokenExpiresin = Integer.Parse(resp("x_refresh_token_expires_in").ToString())
            ExpiresIn = Integer.Parse(resp("expires_in").ToString())

        Else
            Throw New Exception("Imposible obtener Tokens " & respuesta.StatusCode)
        End If

    End Sub
