    Public Sub envia_presupuesto_hub_spot(id As Integer, fecha_creacion As Date, fecha_cierre As Date, empresa As String, vendedor As String, importe As Double, id_hubspot As Integer)
        Try
            If cnn_aw.State = ConnectionState.Closed Then cnn_aw.Open()
            Dim sqlcom As New SqlCommand("insert into dbo.APM_OFERTAS_HUBSPOT(ID,FECHA_CREACION,FECHA_CIERRE,EMPRESA,VENDEDOR,FI_BETR_NETTO,ID_HUB) 
            values (@ID,@FECHA_CREACION,@FECHA_CIERRE,@EMPRESA,@VENDEDOR,@FI_BETR_NETTO,@ID_HUB)", cnn_aw)
            sqlcom.Parameters.AddWithValue("ID", id)
            sqlcom.Parameters.AddWithValue("FECHA_CREACION", fecha_creacion)
            sqlcom.Parameters.AddWithValue("FECHA_CIERRE", fecha_cierre)
            sqlcom.Parameters.AddWithValue("EMPRESA", empresa)
            sqlcom.Parameters.AddWithValue("VENDEDOR", vendedor)
            sqlcom.Parameters.AddWithValue("FI_BETR_NETTO", importe)
            sqlcom.Parameters.AddWithValue("ID_HUB", id_hubspot)
            sqlcom.ExecuteNonQuery()
            cnn_aw.Close()
        Catch ex As Exception
            cnn_aw.Close()
            MsgBox(ex.Message)
        End Try
    End Sub

    Public Function retorna_estatus_pedido_aw(id_pedido As Integer) As DataTable
        Try
            If cnn_aw.State = ConnectionState.Closed Then cnn_aw.Open()
            Dim sqldat As New SqlDataAdapter("select r.STATUS,r.fecha_403,r.fecha_entrega,DATEDIFF (hour,r.fecha_502,r.fecha_entrega) as atraso  from (select a.STATUS,SYSADM.DVD_FUN_GEN_FECHA_PROBLABLE_ENTREGA(a.ID) as fecha_entrega,(select max(x.DATUM)  from SYSADM.BW_AUFTR_HIST x where x.ID =a.ID and x.STATUS_ID =403) as fecha_403,
            coalesce((select max(x.DATUM)  from SYSADM.BW_AUFTR_HIST x where x.ID =a.ID and x.STATUS_ID =520),getdate()) as fecha_502     from SYSADM.BW_AUFTR_KOPF a

            sqldat.Fill(tab)
            cnn_aw.Close()
            Return tab

    Public Function consulta_estatus_liberado(ByVal folio As Integer, ByVal pos As Integer, ByVal reg_point As Integer, ByVal num_piezas As Integer) As Integer
        Try
            If cnn_aw.State = ConnectionState.Open Then cnn_aw.Close()
            cnn_aw.Open()
            Dim sql As String = "SELECT SUM(A.ANZAHL) FROM MEGALUMINIO.SYSADM.PD_AWBAR A WHERE A.ID='" & folio & "' AND A.POINT='" & reg_point & "' and A.POS_NR='" & pos & "'"
            If reg_point = 11 Then sql = "SELECT SUM(A.ANZAHL) FROM MEGALUMINIO.SYSADM.PD_AWBAR A WHERE A.ID='" & folio & "' AND A.POINT in (11,24) and A.POS_NR='" & pos & "'"
            If reg_point = 14 Then sql = "SELECT SUM(A.ANZAHL) FROM MEGALUMINIO.SYSADM.PD_AWBAR A WHERE A.ID='" & folio & "' AND A.POINT in (14,21,29) and A.POS_NR='" & pos & "'"
            If reg_point = 16 Then sql = "SELECT SUM(A.ANZAHL) FROM MEGALUMINIO.SYSADM.PD_AWBAR A WHERE A.ID='" & folio & "' AND A.POINT in (16,30,31) and A.POS_NR='" & pos & "'"
            Dim fbdat As New SqlDataAdapter(sql, cnn_aw)
            Dim tab As New DataTable
            fbdat.Fill(tab)
            If Not IsDBNull(tab.Rows(0).Item(0)) Then
                If tab.Rows(0).Item(0) > num_piezas Then
                    Return num_piezas
                Else
                    Return tab.Rows(0).Item(0)
                End If
            Else
                Return 0
            End If
            If Not IsDBNull(tab.Rows(0).Item(0)) Then
                Return tab.Rows(0).Item(0)
            Else
                Return 0
            End If
            cnn_aw.Close()
        Catch ex As Exception
            MsgBox("Error en posicion: " & pos & " " & ex.Message)
            Return 0
        End Try
    End Function

    Public Function retorna_fecha_prob_entrega_pedido(ByVal pedido As Integer) As String
        Try
            If cnn_aw.State = ConnectionState.Closed Then cnn_aw.Open()
            Dim sqldat As New SqlDataAdapter("SELECT SYSADM.DVD_FUN_GEN_FECHA_PROBLABLE_ENTREGA ('" & pedido & "') ", cnn_aw)
            Dim tab As New DataTable
            sqldat.Fill(tab)
            If tab.Rows.Count > 0 Then
                Try
                    Dim f520 As DateTime = retorna_fecha_estatus_520(pedido)
                    If DateDiff(DateInterval.Hour, tab.Rows(0).Item(0), f520) > 0 Then
                        Dialog26.paner_semaforo.BackColor = Color.Red
                    Else
                        Dialog26.paner_semaforo.BackColor = Color.Green
                    End If
                    Dialog26.txt_atraso.Text = DateDiff(DateInterval.Hour, tab.Rows(0).Item(0), f520) & " Hrs"
                Catch ex As Exception
                End Try
                Return tab.Rows(0).Item(0)
            Else
                Return ""
            End If

        Catch ex As Exception
            Return ""
        End Try
    End Function
