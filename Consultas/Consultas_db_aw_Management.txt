Imports FirebirdSql.Data.FirebirdClient

Module Consultas_db_aw_Management
    Public Function retorna_parametros_sug_megaglass() As DataTable
        Try
            Dim fbdat As New FbDataAdapter("select a.parametro_id,a.clave,a.descripcion,a.signo,a.valor from parametros_costos_sug A order by a.clave", cnn_prod_aw)
            Dim tab As New DataTable
            fbdat.Fill(tab)
            Return tab
        Catch ex As Exception
            MsgBox(ex.Message)
            Return Nothing
        End Try
    End Function
