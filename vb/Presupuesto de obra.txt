    Public Function retorna_datos_documento_detalle(tipo As String, folio As String) As DataTable
        Try

            Dim imp_tot As Double = 0
            If tipo = "CM" Then
                Dim tab As New DataTable
                tab = retorna_datos_documento_ventas_microsip(folio, "C")
                If tab.Rows.Count > 0 Then
                    Return tab
                Else
                    MsgBox("El folio:" & folio & " no existe en el sistema Microsip o es incorrecto", MsgBoxStyle.Information)
                    Return Nothing
                End If
            ElseIf tipo = "ST" Then
                Dim tab As New DataTable
                tab = retorna_datos_sugerido_templado(folio)
                If tab.Rows.Count > 0 Then
                    Return tab
                Else
                    MsgBox("El folio:" & folio & " no existe como sugerido de Templado o es incorrecto", MsgBoxStyle.Information)
                    Return Nothing
                End If
            ElseIf tipo = "SD" Then
                Dim tab As New DataTable
                Dim prec_mq As Double = 0
                tab = retorna_datos_sugerido_doble_vidrio(folio)
                If tab.Rows.Count > 0 Then
                    Return tab
                Else
                    MsgBox("El folio:" & folio & " no existe como sugerido de Insulado o es incorrecto", MsgBoxStyle.Information)
                    Return Nothing
                End If
            ElseIf tipo = "SL" Then
                Dim tab As New DataTable
                Dim prec_mq As Double = 0
                tab = retorna_datos_sugerido_doble_vidrio(folio)
                If tab.Rows.Count > 0 Then
                    Return tab
                Else
                    MsgBox("El folio:" & folio & " no existe como sugerido de Laminado o es incorrecto", MsgBoxStyle.Information)
                    Return Nothing
                End If
            ElseIf tipo = "SF" Then
                'Retorna datos sugerido de Feltes
                Dim tab As New DataTable
                Dim prec_mq As Double = 0
                tab = retorna_datos_sugerido_fletes(folio)
                If tab.Rows.Count > 0 Then
                    Return tab
                Else
                    MsgBox("El folio:" & folio & " no existe como sugerido de laminado o es incorrecto", MsgBoxStyle.Information)
                    Return Nothing
                End If
            Else
                Return Nothing
            End If
        Catch ex As Exception
            MsgBox("Error:retorna_datos_documento_detalle " & ex.Message, MsgBoxStyle.Critical)
            Return Nothing
        End Try
    End Function

    Public Sub calcula_totales_grid()
        Try
            Dim tot_mat_prima As Double = 0
            For Each x As DataGridViewRow In grid_mat_prima.Rows
                tot_mat_prima += x.Cells(3).Value
            Next
            txt_tot_mat_prima.Text = FormatCurrency(tot_mat_prima)
            Dim tot_mano_obra As Double = 0
            For Each x As DataGridViewRow In grid_mano_obra.Rows
                tot_mano_obra += x.Cells(3).Value
            Next
            txt_tot_man_obra.Text = FormatCurrency(tot_mano_obra)
            Dim tot_gas_oper As Double = 0
            For Each x As DataGridViewRow In grid_gas_operac.Rows
                tot_gas_oper += x.Cells(3).Value
            Next
            txt_tot_gas_oper.Text = FormatCurrency(tot_gas_oper)
            txt_cos_primo.Text = FormatCurrency(tot_mat_prima + tot_mano_obra)
            txt_util_neta.Text = FormatCurrency(((tot_mat_prima + tot_mano_obra + tot_gas_oper) / ((100 - txt_util_por.Text) * 0.01)) - (tot_mat_prima + tot_mano_obra + tot_gas_oper))
            txt_subtotal.Text = FormatCurrency(CDbl(txt_cos_primo.Text) + CDbl(txt_util_neta.Text) + CDbl(txt_tot_gas_oper.Text))
            txt_iva.Text = FormatCurrency(CDbl(txt_subtotal.Text) * 0.16)
            txt_total.Text = FormatCurrency(CDbl(txt_subtotal.Text) + CDbl(txt_iva.Text))
            'Se calcula los conceptos que son automaticos
            calcula_conceptos_calc_automatico()
            'se genera de nuevo la sumatoria considerando el cambio de los conceptos automaticos
            tot_mat_prima = 0
            For Each x As DataGridViewRow In grid_mat_prima.Rows
                tot_mat_prima += x.Cells(3).Value
            Next
            txt_tot_mat_prima.Text = FormatCurrency(tot_mat_prima)
            tot_mano_obra = 0
            For Each x As DataGridViewRow In grid_mano_obra.Rows
                tot_mano_obra += x.Cells(3).Value
            Next
            txt_tot_man_obra.Text = FormatCurrency(tot_mano_obra)
            tot_gas_oper = 0
            For Each x As DataGridViewRow In grid_gas_operac.Rows
                tot_gas_oper += x.Cells(3).Value
            Next
            txt_tot_gas_oper.Text = FormatCurrency(tot_gas_oper)
            txt_cos_primo.Text = FormatCurrency(tot_mat_prima + tot_mano_obra)
            txt_util_neta.Text = FormatCurrency(((tot_mat_prima + tot_mano_obra + tot_gas_oper) / ((100 - txt_util_por.Text) * 0.01)) - (tot_mat_prima + tot_mano_obra + tot_gas_oper))
            txt_subtotal.Text = FormatCurrency(CDbl(txt_cos_primo.Text) + CDbl(txt_util_neta.Text) + CDbl(txt_tot_gas_oper.Text))
            txt_iva.Text = FormatCurrency(CDbl(txt_subtotal.Text) * 0.16)
            txt_total.Text = FormatCurrency(CDbl(txt_subtotal.Text) + CDbl(txt_iva.Text))
            If CDbl(txt_subtotal.Text) > 0 Then
                For Each x As DataGridViewRow In grid_mat_prima.Rows
                    x.Cells(6).Value = (x.Cells(3).Value / CDbl(txt_subtotal.Text)) * 100
                Next
                For Each x As DataGridViewRow In grid_mano_obra.Rows
                    x.Cells(6).Value = (x.Cells(3).Value / CDbl(txt_subtotal.Text)) * 100
                Next
                For Each x As DataGridViewRow In grid_gas_operac.Rows
                    x.Cells(6).Value = (x.Cells(3).Value / CDbl(txt_subtotal.Text)) * 100
                Next
            End If
        Catch ex As Exception
            MsgBox("Error:calcula_totales_grid " & ex.Message, MsgBoxStyle.Critical)
        End Try
    End Sub

    Public Function retorna_base_para_calculo_gasto_administrativo_matprima() As Double
        Dim val_ret As Double = 0
        Try
            For Each xx As DataGridViewRow In grid_mat_prima.Rows
                Dim val_mul As Double = retorna_es_base_para_calculo_gasto_administrativo(xx.Cells(2).Value)
                val_ret += val_mul * (xx.Cells(3).Value)
            Next
            For Each xx As DataGridViewRow In grid_mano_obra.Rows
                If retorna_es_base_para_calculo_mano_obra_gasto_administrativo(xx.Cells(2).Value) > 0 Then val_ret += (xx.Cells(3).Value)
            Next
            'MsgBox(val_ret)
            Return val_ret
        Catch ex As Exception
            MsgBox("Error:retorna_base_para_calculo_gasto_administrativo_matprima " & ex.Message, MsgBoxStyle.Critical)
            Return val_ret
        End Try
    End Function
    Public Sub calcula_conceptos_calc_automatico()
        Try
            If BAN_MOD = 0 Then
                'calcula si el concepto es calculo automatico materia prima
                For Each xx As DataGridViewRow In grid_mat_prima.Rows
                    Dim val_autom As Double = retorna_valor_calculo_automatico("conceptos_mat_prima_obra", "a.concepto_mat_prima_id", CDbl(txt_subtotal.Text), xx.Cells(2).Value)
                    If val_autom > 0 Then
                        xx.Cells(3).Value = val_autom
                        xx.Cells(3).ReadOnly = True
                    End If
                Next
                'calcula si el concepto es de calculo automatico para gastos de operacion
                For Each xy As DataGridViewRow In grid_gas_operac.Rows
                    If xy.Cells(2).Value = 3592620 Or xy.Cells(2).Value = 4051235 Or xy.Cells(2).Value = 4186327 Then
                        Dim base_calculo = retorna_base_para_calculo_gasto_administrativo_matprima()
                        Dim val_autom As Double = retorna_valor_calculo_automatico("CONCEPTOS_GAS_OPERACION_OBRA", "a.CONCEPTO_GAS_OPERACION_OBRA_ID", CDbl(base_calculo), xy.Cells(2).Value)
                        If val_autom > 0 Then
                            xy.Cells(3).Value = val_autom
                            xy.Cells(3).ReadOnly = True
                        End If
                    Else
                        Dim val_autom As Double = retorna_valor_calculo_automatico("CONCEPTOS_GAS_OPERACION_OBRA", "a.CONCEPTO_GAS_OPERACION_OBRA_ID", CDbl(txt_subtotal.Text), xy.Cells(2).Value)
                        If val_autom > 0 Then
                            xy.Cells(3).Value = val_autom
                            xy.Cells(3).ReadOnly = True
                        End If
                    End If
                Next
                calcula_comisiones_sin_cambio_totales(txt_subtotal.Text)
            End If
        Catch ex As Exception
            MsgBox("Error:calcula_conceptos_calc_automatico " & ex.Message, MsgBoxStyle.Critical)
        End Try
    End Sub

    Public Sub calcula_conceptos_anexo7(subtotal As Double)
        Try
            For Each x As DataGridViewRow In grid_anexo7.Rows
                Dim tot_pres As Double = CDbl(subtotal)
                Dim tot1, tot2 As Double
                If IsNumeric(x.Cells(3).Value) Then tot1 = tot_pres * x.Cells(3).Value * 0.01
                If IsNumeric(x.Cells(4).Value) Then tot2 = tot1 * x.Cells(4).Value * 0.01
                x.Cells(5).Value = tot2
                Dim ban As Integer = 0
                For Each y As DataGridViewRow In grid_gas_operac.Rows
                    If y.Cells(2).Value = x.Cells(1).Value Then
                        y.Cells(3).Value = tot2
                        ban = 1
                    End If
                Next
                If ban = 0 Then
                    Dim val As Integer = CInt(x.Cells(1).Value)
                    grid_gas_operac.Rows.Add("-1", 0, val, tot2)
                End If
            Next
            'Calcula el gasto de operaciones
            Dim tot_gas_oper As Double = 0
            For Each x As DataGridViewRow In grid_gas_operac.Rows
                tot_gas_oper += x.Cells(3).Value
            Next
            txt_tot_gas_oper.Text = FormatCurrency(tot_gas_oper)
        Catch ex As Exception
            MsgBox("Error:calcula_conceptos_anexo7 " & ex.Message, MsgBoxStyle.Critical)
        End Try
    End Sub

    Public Function valida_sugerido_no_este_asignado(sugerido_id As Integer)
        Dim ret As Integer = 0
        Try
            For Each x As DataGridViewRow In grid_sug_rel.Rows
                If x.Cells(1).Value = sugerido_id Then ret += 1
            Next

        Catch ex As Exception
            Return 0
        End Try
    End Function
