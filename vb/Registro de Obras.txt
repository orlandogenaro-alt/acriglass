    Public Function guarda_obra() As Integer
        Try
            Dim res As String = ""
            If Len(txt_nombre.Text) = 0 Then res = "No se a indicado el nombre de la Obra"
            If Len(txt_direccion.Text) = 0 Then res = "No se a indicado la direccion de la Obra"
            If Len(txt_contacto.Text) = 0 Then res = "El nombre del contacto no fue indicado"
            If grid_responsables.Rows.Count = 0 Then res = "No se a seleccionado un responsable de Obra"
            If Len(res) = 0 Then
                REGISTRA_OBRAS(obra_id, sugerido_id_rel, txt_nombre.Text, txt_direccion.Text, txt_contacto.Text, txt_notas.Text, cal_ini.Value, cal_fin.Value, cmb_estatus.Text, grid_responsables, convierte_img_to_array(pic_img.Image), txt_gas_admin_obr.Text, lbl_avance.Text)
                actualiza_estatus_sugerido_obra(sugerido_id_rel, "Aceptado")
                Form46.lbl_estatus.Text = "Aceptado"
                If cmb_estatus.Text = "Finalizacion" Then
                    actualiza_estatus_obra(obra_id, "Finalizacion")
                    lbl_estatus.Text = "Finalizacion"
                End If
                Return 1
            Else
                MsgBox(res, MsgBoxStyle.Information)
                Return 0
            End If
        Catch ex As Exception
            MsgBox("Error:guarda_obra " & ex.Message, MsgBoxStyle.Critical)
            Return 0
        End Try
    End Function

    Public Sub genera_total_grid()

    Public Function genera_salida_almacen_prod_proceso_mat_obra(obra_id As Integer) As Integer
        Try
            Dim tab As New DataTable
            tab = retorna_doctos_salidas_por_obra_det(obra_id, 37309)
            Dim docto_salida_id_pt As Integer = 0
            If tab.Rows.Count > 0 Then
                Dim tab1, tab2 As New DataTable
                tab1 = consulta_parametros_configuracion_saca("Concepto salida Prod.proceso", "PT")
                Dim con_sal As Integer = 0
                If tab1.Rows.Count > 0 Then con_sal = tab1.Rows(0).Item(0)
                tab2 = consulta_parametros_configuracion_saca("Almacen de Prod.proceso", "AT")
                Dim alm_pp As Integer = 0
                If tab2.Rows.Count > 0 Then alm_pp = tab2.Rows(0).Item(0)
                Dim docto_id_sal As Integer = 0
                'Realizamos la salida de la materia prima del almacen de produccion en proceso            
                If con_sal > 0 And alm_pp > 0 Then
                    docto_id_sal = valida_no_exista_mov_microsip_sali(obra_id, con_sal, 0, "F", cal_fin.Value, obra_id)
                    If docto_id_sal = 0 Then docto_id_sal = genera_mov_salida_proceso_por_mts(alm_pp, 0, obra_id, con_sal, tab, "S", txt_nombre.Text & " Salida de material de produccion en proceso de la Obra", tab2.Rows(0).Item(1), obra_id, "F", obra_id, 0, 0)
                    If docto_id_sal > 0 Then
                        'Genera entrada de producto Terminado
                        Dim docto_entrad_id As Integer = 0
                        Dim tab11, tab22 As New DataTable
                        tab11 = consulta_parametros_configuracion_saca("Concepto Entrada Producto Terminado", "ET")
                        Dim con_ept As Integer = 0
                        If tab11.Rows.Count > 0 Then con_ept = tab11.Rows(0).Item(0)
                        tab22 = consulta_parametros_configuracion_saca("Almacen de Producto Terminado", "AP")
                        Dim alm_ept As Integer = 0
                        If tab22.Rows.Count > 0 Then alm_ept = tab22.Rows(0).Item(0)
                        docto_entrad_id = valida_no_exista_mov_microsip_entrada(obra_id, con_ept, 0, "F", cal_fin.Value, obra_id)
                        Dim tab5 As New DataTable
                        tab5 = consulta_parametros_configuracion_saca("Articulo de Producto Terminado", "AR")
                        Dim art_id_pt As Integer = 0
                        If tab5.Rows.Count > 0 Then art_id_pt = tab5.Rows(0).Item(0)
                        Dim tab_art As New DataTable
                        If art_id_pt > 0 Then
                            'realizamos la entrada del producto terminado
                            Dim cost_prod As Double = Consultas_microsip.consulta_costo_tot_mov_microsip(docto_id_sal)
                            tab_art.Columns.Add("articulo_id", GetType(Integer))
                            tab_art.Columns.Add("uds", GetType(Double))
                            tab_art.Columns.Add("costo_unit", GetType(Double))
                            tab_art.Rows.Add(art_id_pt, 1, cost_prod)
                        End If
                        If docto_entrad_id = 0 Then
                            docto_entrad_id = genera_entrada_salida_producto_terminado_saca(obra_id, alm_ept, con_ept, "E", txt_nombre.Text & " Entrada de producto terminado de Obra ", tab_art)
                        End If
                        If docto_entrad_id > 0 Then
                            'Realizamos la salida del producto terminado
                            Dim tab13 As New DataTable
                            tab13 = consulta_parametros_configuracion_saca("Concepto Salida Producto Terminado", "ST")
                            Dim con_spt As Integer = 0
                            If tab13.Rows.Count > 0 Then con_spt = tab13.Rows(0).Item(0)
                            If con_spt > 0 Then
                                docto_salida_id_pt = valida_no_exista_mov_microsip_sali(obra_id, con_spt, 0, "F", cal_fin.Value, obra_id)
                                If docto_salida_id_pt = 0 Then
                                    docto_salida_id_pt = genera_entrada_salida_producto_terminado_saca(obra_id, alm_ept, con_spt, "S", txt_nombre.Text & " Salida del Costo de producto terminado de Obra ", tab_art)
                                End If
                            End If
                        End If
                    End If
                End If
            End If
            Return docto_salida_id_pt
        Catch ex As Exception
            MsgBox("Error:genera_salida_almacen_prod_proceso_mat_obra  " & ex.Message, MsgBoxStyle.Critical)
            Return 0
        End Try
    End Function
    Public Sub cerrar_obra()
        Try
            If Trim(lbl_estatus.Text) = "Obra Terminada" Then
                If (MsgBox("Al cerrar la Obra, todas las ordenes que se encuentran en produccion se enviaran a Fin de Produccion, las que no se liberan a produccion se estaran Cancelando, No sera posible agregar mas ordenes de trabajo o materiales, ¿Deseas Continuar? ", MsgBoxStyle.OkCancel Or MsgBoxStyle.Question) = vbOK) Then
                    Dim tab As New DataTable
                    tab = retorna_ordenes_por_obra(obra_id, "C")
                    For Each x As DataRow In tab.Rows
                        If x("estatus_2") = "P" Then
                            actualiza_estatus_orden_obra(x("docto_orden_obra_id"), "F")
                            guarda_bitacora_costos_prod(x("docto_orden_obra_id"), x("docto_orden_obra_id"), usuario_gl, "Estatus: " & "Fin Produccion: Por cierre de Obra ")
                        ElseIf x("estatus_2") = "O" Then
                            actualiza_estatus_orden_obra(x("docto_orden_obra_id"), "C")
                            guarda_bitacora_costos_prod(x("docto_orden_obra_id"), x("docto_orden_obra_id"), usuario_gl, "Estatus: " & "Cancelado: Por cierre de Obra ")
                        End If
                    Next
                    Dim tab2 As New DataTable
                    tab2 = retorna_doctos_salidas_por_obra(obra_id)
                    For Each x As DataRow In tab2.Rows
                        If Trim(x("estatus")) = "Pendiente" Then
                            actualiza_movimiento_solicitud_material(x("docto_solicitud_id"), 0, "Cancelado", x("folio_microsip"))
                            guarda_bitacora_costos_prod(x("docto_solicitud_id"), x("docto_solicitud_id"), usr_gl, "Documento Cancelado")
                        End If
                    Next
                    'Genera los movimientos en sistema Microsip
                    If genera_salida_almacen_prod_proceso_mat_obra(obra_id) > 0 Then
                        MsgBox("Los movimientos en el sistema Microsip se realizaron Correctamente!!", MsgBoxStyle.Information)
                    End If
                    actualiza_estatus_obra(obra_id, "Finalizacion")
                    lbl_estatus.Text = "Finalizacion"
                    cmb_estatus.Text = "Finalizacion"
                End If
            Else
                MsgBox("Para finalizar la obra, es necesario que su estatus sea Obra Terminada. Actualice el estatus para poder continuar.", MsgBoxStyle.Information)
            End If
        Catch ex As Exception
            MsgBox("Error:cerrar_obra  " & ex.Message, MsgBoxStyle.Critical)
        End Try
    End Sub
